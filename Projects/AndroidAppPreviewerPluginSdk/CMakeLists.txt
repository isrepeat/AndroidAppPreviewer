cmake_minimum_required(VERSION 3.22)
project(AndroidAppPreviewerPluginSdk LANGUAGES CXX)
get_filename_component(previewerRepositoryRoot "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
if(NOT ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT)
    message(FATAL_ERROR "Set ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT to the directory of restored UDK NuGet packages.")
endif()
include("${previewerRepositoryRoot}/Packages/Windows/AndroidAppPreviewerPluginSdk/Delivery/CMake/UdkNuGetDependencies.cmake")
android_app_previewer_import_udk("${previewerRepositoryRoot}/Packages/Windows/AndroidAppPreviewerPluginSdk/Delivery/NuGet/Delivery.json" "${ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Files.json" component)
string(JSON count LENGTH "${component}" sources)
math(EXPR last "${count} - 1")
set(sources)
foreach(index RANGE ${last})
    string(JSON source GET "${component}" sources ${index})
    list(APPEND sources "${previewerRepositoryRoot}/${source}")
endforeach()
add_library(AndroidAppPreviewerPluginSdk STATIC ${sources})
target_compile_features(AndroidAppPreviewerPluginSdk PUBLIC cxx_std_20)
target_include_directories(AndroidAppPreviewerPluginSdk PUBLIC "${previewerRepositoryRoot}/Projects" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(AndroidAppPreviewerPluginSdk PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AndroidAppPreviewerPluginSdk PRIVATE /utf-8 /EHsc)
target_link_libraries(AndroidAppPreviewerPluginSdk PUBLIC UniversalDevKit::XamlRuntime.Static UniversalDevKit::Diagnostic.Dynamic PRIVATE UniversalDevKit::OpenGLESRenderer.Static UniversalDevKit::ANGLE)
