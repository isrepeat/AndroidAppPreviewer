function(previewerFilePatternRegex pattern output)
    set(regex "${pattern}")
    foreach(character IN ITEMS "." "+" "^" "$" "(" ")" "[" "]" "{" "}" "|")
        string(REPLACE "${character}" "\\${character}" regex "${regex}")
    endforeach()
    string(REPLACE "**/" "@RECURSIVE@" regex "${regex}")
    string(REPLACE "**" "@ALL@" regex "${regex}")
    string(REPLACE "*" "[^/]*" regex "${regex}")
    string(REPLACE "?" "[^/]" regex "${regex}")
    string(REPLACE "@RECURSIVE@" "(.*/)?" regex "${regex}")
    string(REPLACE "@ALL@" ".*" regex "${regex}")
    set(${output} "^${regex}$" PARENT_SCOPE)
endfunction()

function(previewerResolveFileSelection json field excludeField output)
    string(JSON type TYPE "${json}" ${field})
    if(NOT type STREQUAL "ARRAY")
        message(FATAL_ERROR "${field} must be an array")
    endif()
    string(JSON count LENGTH "${json}" ${field})
    set(files)
    if(count GREATER 0)
        math(EXPR last "${count} - 1")
        foreach(index RANGE ${last})
            string(JSON type TYPE "${json}" ${field} ${index})
            string(JSON pattern GET "${json}" ${field} ${index})
            if(NOT type STREQUAL "STRING")
                message(FATAL_ERROR "${field} must contain strings")
            endif()
            string(REGEX REPLACE "[*?]" "_" checkedPath "${pattern}")
            if(IS_ABSOLUTE "${checkedPath}" OR checkedPath MATCHES "(^|/)[.][.](/|$)")
                message(FATAL_ERROR "Invalid source pattern: ${pattern}")
            endif()
            if(pattern MATCHES "[*?]")
                string(REGEX REPLACE "[*?].*$" "" prefix "${pattern}")
                string(REGEX REPLACE "[^/]+$" "" directory "${prefix}")
                file(GLOB_RECURSE candidates LIST_DIRECTORIES false CONFIGURE_DEPENDS
                    RELATIVE "${previewerRepositoryRoot}" "${previewerRepositoryRoot}/${directory}*")
                previewerFilePatternRegex("${pattern}" regex)
                foreach(candidate IN LISTS candidates)
                    if(candidate MATCHES "${regex}" AND NOT candidate MATCHES "/(\\.git|!artifacts|!tools|bin|obj)/")
                        list(APPEND files "${candidate}")
                    endif()
                endforeach()
            else()
                if(pattern IN_LIST files)
                    message(FATAL_ERROR "Duplicate component file: ${pattern}")
                endif()
                list(APPEND files "${pattern}")
            endif()
        endforeach()
    endif()
    string(JSON excludeType ERROR_VARIABLE missing TYPE "${json}" ${excludeField})
    if(NOT missing)
        if(NOT excludeType STREQUAL "ARRAY")
            message(FATAL_ERROR "${excludeField} must be an array")
        endif()
        string(JSON excludeCount LENGTH "${json}" ${excludeField})
        if(excludeCount GREATER 0)
            math(EXPR last "${excludeCount} - 1")
            foreach(index RANGE ${last})
                string(JSON type TYPE "${json}" ${excludeField} ${index})
                string(JSON pattern GET "${json}" ${excludeField} ${index})
                if(NOT type STREQUAL "STRING")
                    message(FATAL_ERROR "${excludeField} must contain strings")
                endif()
                string(REGEX REPLACE "[*?]" "_" checkedPath "${pattern}")
                if(IS_ABSOLUTE "${checkedPath}" OR checkedPath MATCHES "(^|/)[.][.](/|$)")
                    message(FATAL_ERROR "Invalid source pattern: ${pattern}")
                endif()
                previewerFilePatternRegex("${pattern}" regex)
                list(FILTER files EXCLUDE REGEX "${regex}")
            endforeach()
        endif()
    endif()
    list(REMOVE_DUPLICATES files)
    set(${output} "${files}" PARENT_SCOPE)
endfunction()

    COMMAND "${ANDROID_APP_PREVIEWER_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${reader}" -Path "${componentPath}" -NativeSources
    OUTPUT_VARIABLE nativeProject
    ERROR_VARIABLE error
    RESULT_VARIABLE result
    ENCODING UTF-8
)
if(NOT result STREQUAL "0")
    message(FATAL_ERROR "Cannot read PluginSDK source selection: ${error}")
endif()
previewerResolveFileSelection("${nativeProject}" sources excludeSources sources)

add_library(AndroidAppPreviewerPluginSdk STATIC ${sources})
