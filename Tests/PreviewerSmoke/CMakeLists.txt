cmake_minimum_required(VERSION 3.22)
project(PreviewerSmoke LANGUAGES CXX)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
get_filename_component(previewerRepositoryRoot "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT)
    message(FATAL_ERROR "Set ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT to restored UDK NuGet packages.")
endif()
include("${previewerRepositoryRoot}/Packages/Windows/AndroidAppPreviewerPluginSdk/Delivery/CMake/UdkNuGetDependencies.cmake")
android_app_previewer_import_udk("${previewerRepositoryRoot}/Packages/Windows/AndroidAppPreviewerPluginSdk/Delivery/NuGet/Delivery.xml" "${ANDROID_APP_PREVIEWER_UDK_PACKAGES_ROOT}")
add_executable(PreviewerSmoke PreviewerSmoke.cpp)
target_compile_features(PreviewerSmoke PRIVATE cxx_std_20)
target_link_libraries(PreviewerSmoke PRIVATE AndroidAppPreviewerPlugin::UdkHeaders)
target_include_directories(PreviewerSmoke PRIVATE "${previewerRepositoryRoot}/Sources")
target_compile_options(PreviewerSmoke PRIVATE /utf-8 /EHsc)
target_compile_definitions(PreviewerSmoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
